\section*{Formula Ringkas}
\begin{multicols}{2}
\footnotesize
\begin{formula}
\textbf{PDB dan solusi}
\[
a_ny^{(n)}+\cdots+a_1y'+a_0y=f(x).
\]
Suatu $y$ adalah solusi bila substitusi ke persamaan menghasilkan identitas. PDB linear orde $n$ memiliki solusi umum yang memuat $n$ konstanta bebas; kondisi awal/batas memilih solusi partikular.
\end{formula}
\begin{formula}
\textbf{Laplace}
\[
\Lap{f(t)}=F(s)=\int_0^\infty e^{-st}f(t)\,dt.
\]
\[
\Lap{af+bg}=aF+bG.
\]
\[
\Lap{y'}=sY-y(0),\qquad
\Lap{y''}=s^2Y-sy(0)-y'(0).
\]
\end{formula}
\begin{formula}
\textbf{Tabel inti Laplace}
\[
\Lap{1}=\frac1s,\quad \Lap{t^n}=\frac{n!}{s^{n+1}},\quad
\Lap{e^{at}}=\frac1{s-a},
\]
\[
\Lap{\sin at}=\frac{a}{s^2+a^2},\qquad
\Lap{\cos at}=\frac{s}{s^2+a^2}.
\]
\end{formula}
\begin{formula}
\textbf{Shifting}
Jika $F(s)=\Lap{f(t)}$,
\[
\Lap{e^{-at}f(t)}=F(s+a),
\]
\[
\Lap{f(t-a)u(t-a)}=e^{-as}F(s).
\]
\end{formula}
\begin{formula}
\textbf{Deret Fourier periode $2L$}
\[
f(x)\sim \frac{a_0}{2}+\sum_{n=1}^{\infty}\left[a_n\cos\frac{n\pi x}{L}+b_n\sin\frac{n\pi x}{L}\right],
\]
\[
a_n=\frac1L\int_{-L}^{L}f(x)\cos\frac{n\pi x}{L}\,dx,
\quad
b_n=\frac1L\int_{-L}^{L}f(x)\sin\frac{n\pi x}{L}\,dx.
\]
\end{formula}
\begin{formula}
\textbf{Even/odd}
Jika $f$ genap, $b_n=0$. Jika $f$ ganjil, $a_0=a_n=0$.
\[
\text{rata-rata }f=\frac1{b-a}\int_a^b f(x)\,dx.
\]
\end{formula}
\begin{formula}
\textbf{BVP dan PDE}
Persamaan Laplace:
\[
T_{xx}+T_{yy}=0.
\]
Wave equation 1D:
\[
u_{tt}=c^2u_{xx}.
\]
Pemisahan variabel: $T(x,y)=X(x)Y(y)$ atau $u(x,t)=X(x)G(t)$.
\end{formula}
\begin{formula}
\textbf{Tiga unsur BVP}
\begin{enumerate}
\item governing equation;
\item domain;
\item boundary conditions.
\end{enumerate}
Prosedur umum: separation of variables $\to$ seleksi dengan kondisi batas $\to$ Fourier untuk menyesuaikan data batas/awal.
\end{formula}
\end{multicols}

\section*{M1: Refresh PDB dan Kalkulus}
{\color{iteraGray}\small Fondasi sebelum Transformasi Laplace.}\par\vspace{1mm}
\begin{multicols}{2}
\begin{qbox}[title={M1.1: Klasifikasi persamaan diferensial\hfill \level{L1}\;\src{Boas}}]
\small Untuk setiap persamaan, tentukan: PDB/PDP, orde, dan linear/nonlinear.
\begin{enumerate}
\item $y'+xy^2=1$.
\item $xy'+y=e^x$.
\item $u_t=c^2u_{xx}$.
\item $y''+4y'+5y=\sin x$.
\end{enumerate}
\end{qbox}
\begin{qbox}[title={M1.2: Verifikasi solusi\hfill \level{L1}\;\src{Boas}}]
\small Verifikasikan dengan substitusi bahwa $y=\sin x+C$ memenuhi $y'=\cos x$. Lalu verifikasikan bahwa $y=Ae^x+Be^{-x}$ memenuhi $y''=y$. Jelaskan mengapa konstanta bebas tidak mengganggu identitas.
\end{qbox}
\begin{qbox}[title={M1.3: Solusi umum dan solusi partikular\hfill \level{L2}\;\src{Boas}}]
\small Diberikan $y''=y$ dengan solusi umum $y=Ae^x+Be^{-x}$. Tentukan $A,B$ jika $y(0)=2$ dan $y'(0)=0$. Bedakan kalimat ``solusi umum'' dan ``solusi partikular'' pada hasil Anda.
\end{qbox}
\begin{qbox}[title={M1.4: Kondisi awal atau kondisi batas?\hfill \level{L1}\;\src{Var}}]
\small Klasifikasikan setiap informasi sebagai kondisi awal atau kondisi batas: (a) $y(0)=1$, $y'(0)=0$; (b) $T(0)=100$, $T(10)=50$; (c) $u(0,t)=u(L,t)=0$; (d) $u(x,0)=f(x)$. Jelaskan alasan Anda.
\end{qbox}
\begin{qbox}[title={M1.5: Benda jatuh\hfill \level{L2}\;\src{PR}}]
\small Sebuah benda mempunyai $v(0)=1\,\mathrm{m/s}$ dan $x(0)=2\,\mathrm m$. Dengan $dv/dt=g$, tentukan $v(t)$ dan $x(t)$. Hitung posisi pada $t=10$ s. Gunakan $g$ sebagai parameter, lalu evaluasi jika $g=9.8\,\mathrm{m/s^2}$.
\end{qbox}
\begin{qbox}[title={M1.6: Massa-pegas\hfill \level{L2}\;\src{PR}}]
\small Model gerak bebas memenuhi $y''+\omega^2y=0$. Jika $y(0)=0$ dan $y'(0)=1$, tentukan solusi dalam bentuk sinus-kosinus. Verifikasikan kembali kondisi awalnya.
\end{qbox}
\begin{qbox}[title={M1.7: Persamaan separabel\hfill \level{L2}\;\src{Boas}}]
\small Selesaikan $dN/dt=-\lambda N$ dengan $N(0)=N_0$. Tunjukkan langkah pemisahan variabel, integral, penentuan konstanta, dan interpretasikan tanda negatif pada laju perubahan.
\end{qbox}
\begin{qbox}[title={M1.8: Integral sebagai penyelesaian PDB\hfill \level{L2}\;\src{Boas}}]
\small Jelaskan mengapa menghitung $y=\int f(x)\,dx$ sama dengan menyelesaikan PDB $y'=f(x)$. Gunakan contoh $f(x)=3x^2-2x+1$ dan tuliskan keluarga solusinya.
\end{qbox}
\begin{qbox}[title={M1.9: Refresh turunan dan integral\hfill \level{L2}\;\src{Var}}]
\small Hitung tanpa CAS:
\begin{enumerate}
\item $\frac{d}{dt}(t^2e^{-3t})$;
\item $\frac{d^2}{dt^2}(\sin 2t)$;
\item $\int (4t^3-2t+5)\,dt$;
\item $\int_0^\pi \sin^2 x\,dx$.
\end{enumerate}
\end{qbox}
\begin{qbox}[title={M1.10: Dari model ke persamaan\hfill \level{L3}\;\src{Var}}]
\small Untuk masing-masing fenomena berikut, tuliskan variabel utama, dugaan persamaan pengatur paling sederhana, dan data tambahan yang diperlukan agar solusi unik: pendinginan batang, osilasi pegas, gelombang tali, dan gerak jatuh.
\end{qbox}
\end{multicols}

\section*{M2: Transformasi Laplace I}
{\color{iteraGray}\small Definisi, linearitas, tabel dasar, dan manipulasi parameter.}\par\vspace{1mm}
\begin{multicols}{2}
\begin{qbox}[title={M2.1: Makna Transformasi Laplace\hfill \level{L1}\;\src{PR}}]
\small Jelaskan apa yang dimaksud dengan Transformasi Laplace dan mengapa transformasi ini berguna ketika menyelesaikan PDB linear dengan kondisi awal.
\end{qbox}
\begin{qbox}[title={M2.2: Dari definisi: fungsi konstanta\hfill \level{L2}\;\src{Boas}}]
\small Mulai dari definisi $\Lap{f}=\int_0^\infty e^{-st}f(t)dt$, turunkan $\Lap{1}=1/s$ dan sebutkan syarat pada $s$ agar integral konvergen.
\end{qbox}
\begin{qbox}[title={M2.3: Dari definisi: eksponensial\hfill \level{L2}\;\src{Boas}}]
\small Turunkan langsung $\Lap{e^{-at}}=1/(s+a)$. Jelaskan apa yang berubah jika fungsi menjadi $e^{at}$.
\end{qbox}
\begin{qbox}[title={M2.4: Linearitas\hfill \level{L2}\;\src{Var}}]
\small Gunakan linearitas untuk menghitung $\Lap{4t^2-3\sin 2t+5e^{-t}}$. Pisahkan setiap suku dan tuliskan baris tabel yang digunakan.
\end{qbox}
\begin{qbox}[title={M2.5: Tugas resmi: transformasi dasar\hfill \level{L2}\;\src{PR}}]
\small Tentukan Transformasi Laplace dari: (a) $5t^2$; (b) $3\sin 3t$; (c) $te^{2t}$.
\end{qbox}
\begin{qbox}[title={M2.6: Eksponensial kompleks\hfill \level{L3}\;\src{PR}}]
\small Dengan $e^{iat}=\cos at+i\sin at$, turunkan transformasi $e^{iat}$ dari definisi atau transformasi eksponensial. Pisahkan bagian real dan imajiner untuk memperoleh transformasi sinus dan cosinus.
\end{qbox}
\begin{qbox}[title={M2.7: Kombinasi trigonometrik\hfill \level{L2}\;\src{PR}}]
\small Tentukan transformasi $f(t)=2-2\cos 2t$ dan $f(t)=\frac1t\sin 3t$. Untuk soal kedua, identifikasi entri/relasi tabel yang diperlukan.
\end{qbox}
\begin{qbox}[title={M2.8: Eksponensial kali sinus\hfill \level{L2}\;\src{PR}}]
\small Tentukan $\Lap{e^{2t}\sin 2t}$ dengan shifting theorem. Tuliskan bentuk $F(s-a)$ yang Anda gunakan.
\end{qbox}
\begin{qbox}[title={M2.9: Selisih eksponensial dibagi $t$\hfill \level{L3}\;\src{PR}}]
\small Tentukan transformasi $\displaystyle f(t)=\frac{10}{t}(e^{-t}-e^{-7t})$. Jelaskan relasi tabel atau integral parameter yang digunakan.
\end{qbox}
\begin{qbox}[title={M2.10: Gelombang bergeser\hfill \level{L3}\;\src{PR}}]
\small Anggap $k$ dan $x$ konstanta. Tentukan Transformasi Laplace terhadap $t$ dari $f(t)=2\sin(kx-t)$. Uraikan terlebih dahulu dengan identitas sinus selisih.
\end{qbox}
\begin{qbox}[title={M2.11: Tanh sebagai tantangan\hfill \level{L4}\;\src{PR}}]
\small Selidiki $\Lap{\tanh 2t}$. Jika tidak dapat diperoleh dari tabel inti, jelaskan strategi yang mungkin: identitas eksponensial, tabel yang lebih lengkap, atau pendekatan numerik. Fokus pada proses, bukan hanya hasil akhir.
\end{qbox}
\begin{qbox}[title={M2.12: Turunkan rumus dari parameter\hfill \level{L3}\;\src{Boas}}]
\small Mulai dari $\Lap{e^{-at}}=1/(s+a)$. Diferensiasikan terhadap $s$ untuk memperoleh transformasi $te^{-at}$ dan $t^2e^{-at}$.
\end{qbox}
\end{multicols}

\section*{M3: Transformasi Laplace II}
{\color{iteraGray}\small Transformasi turunan, kondisi awal, invers, shifting, dan pecahan parsial.}\par\vspace{1mm}
\begin{multicols}{2}
\begin{qbox}[title={M3.1: Transformasi turunan pertama\hfill \level{L2}\;\src{Boas}}]
\small Dengan integral parsial, turunkan $\Lap{y'}=sY-y(0)$. Nyatakan jelas pilihan $u$ dan $dv$ serta asumsi pada suku batas ketika $t\to\infty$.
\end{qbox}
\begin{qbox}[title={M3.2: Transformasi turunan kedua\hfill \level{L2}\;\src{Boas}}]
\small Dari rumus $\Lap{y'}$, turunkan $\Lap{y''}=s^2Y-sy(0)-y'(0)$.
\end{qbox}
\begin{qbox}[title={M3.3: Orde tiga dan empat\hfill \level{L2}\;\src{Var}}]
\small Tuliskan $\Lap{y^{(3)}}$ dan $\Lap{y^{(4)}}$ secara lengkap. Kemudian evaluasi jika $y(0)=1$, $y'(0)=-2$, $y''(0)=0$, $y^{(3)}(0)=5$.
\end{qbox}
\begin{qbox}[title={M3.4: Verifikasi entri tabel dengan diferensiasi parameter\hfill \level{L3}\;\src{PR}}]
\small Verifikasikan transformasi $t\cos at$ dengan mendiferensiasikan transformasi $\sin at$ atau $\cos at$ terhadap parameter $a$. Tunjukkan tanda yang muncul dengan teliti.
\end{qbox}
\begin{qbox}[title={M3.5: Shifting eksponensial\hfill \level{L3}\;\src{PR}}]
\small Buktikan bentuk $\Lap{e^{-at}g(t)}=G(s+a)$ dari definisi. Gunakan untuk memperoleh transformasi $e^{-2t}\cos 3t$ dan $e^{-2t}(1-2t)$.
\end{qbox}
\begin{qbox}[title={M3.6: Invers Laplace langsung\hfill \level{L2}\;\src{Boas}}]
\small Tentukan invers dari: (a) $6/s^4$; (b) $5/(s^2+25)$; (c) $s/(s^2+9)$; (d) $1/(s+4)$.
\end{qbox}
\begin{qbox}[title={M3.7: Invers rasional linear\hfill \level{L3}\;\src{Boas}}]
\small Tentukan $\iLap{(5-2s)/(s^2+s-2)}$. Faktorkan penyebut dan gunakan pecahan parsial.
\end{qbox}
\begin{qbox}[title={M3.8: Lengkapi kuadrat\hfill \level{L3}\;\src{Boas}}]
\small Tentukan $\iLap{(6-s)/(s^2+4s+20)}$. Ubah penyebut menjadi $(s+2)^2+16$ dan pecah pembilang agar cocok dengan sinus/kosinus bergeser.
\end{qbox}
\begin{qbox}[title={M3.9: Faktor berulang\hfill \level{L3}\;\src{Var}}]
\small Uraikan $1/[s(s+2)^2]$ menjadi pecahan parsial dan tentukan invers Laplace-nya.
\end{qbox}
\begin{qbox}[title={M3.10: Shifting waktu\hfill \level{L3}\;\src{Boas}}]
\small Tentukan invers $e^{-2s}/s$ dan $e^{-\pi s}s/(s^2+1)$. Tulis jawaban menggunakan fungsi step $u(t-a)$ dan sketsa kapan fungsi mulai aktif.
\end{qbox}
\begin{qbox}[title={M3.11: Pilih strategi sebelum menghitung\hfill \level{L2}\;\src{Var}}]
\small Untuk setiap bentuk, tentukan strategi awal: langsung tabel, pecahan parsial, melengkapi kuadrat, shifting-$s$, atau shifting-waktu: $4/(s^2+16)$, $(s+2)/[(s+2)^2+9]$, $(3s+5)/[s(s+2)]$, $e^{-3s}/(s^2+1)$.
\end{qbox}
\begin{qbox}[title={M3.12: Inverse transform campuran\hfill \level{L4}\;\src{Var}}]
\small Tentukan $\iLap{(2s^2+7s+9)/[s(s+1)(s+3)]}$. Verifikasi hasil akhir dengan mentransformasikan kembali jawaban Anda.
\end{qbox}
\end{multicols}

\section*{M4: PDB dengan Transformasi Laplace}
{\color{iteraGray}\small Penerapan penuh pada IVP, sistem PDB, dan integral.}\par\vspace{1mm}
\begin{multicols}{2}
\begin{qbox}[title={M4.1: IVP orde satu\hfill \level{L2}\;\src{Var}}]
\small Selesaikan dengan Laplace: $y'+y=2$, $y(0)=1$. Tunjukkan transformasi, penyelesaian aljabar untuk $Y(s)$, pecahan parsial, dan invers.
\end{qbox}
\begin{qbox}[title={M4.2: IVP orde dua eksponensial\hfill \level{L3}\;\src{PR}}]
\small Selesaikan $y''-y=8e^{3t}$ dengan $y(0)=1$, $y'(0)=3$ menggunakan Transformasi Laplace.
\end{qbox}

\begin{qbox}[title={M4.4: Resonansi trigonometrik\hfill \level{L3}\;\src{PR}}]

\end{qbox}
\begin{qbox}[title={M4.5: Forcing hiperbolik\hfill \level{L3}\;\src{PR}}]

\begin{qbox}[title={M4.6: Eksponensial-sinus\hfill \level{L3}\;\src{PR}}]

\section*{M5: Dasar Fourier}
{\color{iteraGray}\small Fungsi harmonik, periodik, rata-rata, ortogonalitas, dan parity.}\par\vspace{1mm}
\begin{multicols}{2}
\begin{qbox}[title={M5.1: Fungsi harmonik\hfill \level{L1}\;\src{Var}}]
\small Untuk $y=A\sin(\omega t+\phi)$, jelaskan makna amplitudo, frekuensi sudut, periode, frekuensi, dan fase. Hitung periode jika $\omega=4\pi$.
\end{qbox}
\begin{qbox}[title={M5.2: Radian dan periodisitas\hfill \level{L1}\;\src{Var}}]
\small Ubah $30^\circ,90^\circ,225^\circ$ ke radian. Tentukan periode $\sin 3x$, $\cos(x/2)$, dan $\sin(2\pi t/T)$.
\end{qbox}
\begin{qbox}[title={M5.3: Fungsi periodik atau bukan\hfill \level{L2}\;\src{Var}}]
\small Tentukan apakah fungsi berikut periodik dan, jika ya, periodenya: $\sin x+\cos 2x$, $\sin x+\cos(\sqrt2x)$, $e^{-x}\sin x$, dan fungsi gelombang kotak periode $2L$.
\end{qbox}
\begin{qbox}[title={M5.4: Rata-rata fungsi\hfill \level{L2}\;\src{Var}}]
\small Hitung rata-rata $f(x)=x^2$ pada $[-1,1]$, $\sin^2 x$ pada $[-\pi,\pi]$, dan $\cos x$ pada satu periode penuh.
\end{qbox}
\begin{qbox}[title={M5.5: Ortogonalitas\hfill \level{L2}\;\src{Var}}]
\small Evaluasi $\int_{-\pi}^{\pi}\sin(mx)\sin(nx)dx$ untuk $m=n$ dan $m\ne n$ pada beberapa pasangan bilangan bulat. Apa pola yang terlihat?
\end{qbox}
\begin{qbox}[title={M5.6: Representasi sinus-kosinus\hfill \level{L1}\;\src{PR}}]
\small Jelaskan bagaimana fungsi periodik dapat direpresentasikan sebagai kombinasi sinus dan cosinus. Buat ilustrasi konseptual dengan menyebutkan fungsi target dan beberapa harmoniknya.
\end{qbox}
\begin{qbox}[title={M5.7: Asal koefisien Fourier\hfill \level{L3}\;\src{PR}}]
\small Turunkan ide formula $a_n=(1/\pi)\int_{-\pi}^{\pi}f(x)\cos nx\,dx$ dan $b_n=(1/\pi)\int_{-\pi}^{\pi}f(x)\sin nx\,dx$ dengan memanfaatkan ortogonalitas.
\end{qbox}
\begin{qbox}[title={M5.8: Even dan odd\hfill \level{L2}\;\src{Var}}]
\small Tentukan apakah $x^2$, $x^3$, $\sin x$, $\cos x$, $x\sin x$, dan $x\cos x$ genap atau ganjil. Prediksi koefisien Fourier mana yang otomatis nol.
\end{qbox}
\begin{qbox}[title={M5.9: Gelombang progresif\hfill \level{L2}\;\src{Var}}]
\small Untuk $\eta(x,t)=A\cos(kx-\omega t)$, tentukan panjang gelombang, periode, dan cepat rambat fase. Hubungkan bentuk ini dengan penggunaan Fourier dalam teknik kelautan.
\end{qbox}
\begin{qbox}[title={M5.10: Dari grafik ke fungsi potongan\hfill \level{L3}\;\src{Var}}]
\small Gambarkan satu periode gelombang persegi dengan nilai $-1$ untuk $-\pi<x<0$ dan $1$ untuk $0<x<\pi$. Tulis definisi fungsi potongannya dan tentukan sifat genap/ganjil.
\end{qbox}
\end{multicols}

\begin{qbox}[title={M6.7: Half-range cosine series\hfill \level{L3}\;\src{Var}}]

\section*{M10: Separation of Variables}
{\color{iteraGray}\small Membangun keluarga solusi dan memilih mode yang sesuai kondisi batas.}\par\vspace{1mm}
\begin{multicols}{2}
\begin{qbox}[title={M10.1: Asumsi produk\hfill \level{L2}\;\src{Var}}]
\small Untuk persamaan Laplace $T_{xx}+T_{yy}=0$, ambil $T=X(x)Y(y)$. Substitusikan dan pisahkan sehingga diperoleh $X''/X=-Y''/Y=\lambda$.
\end{qbox}
\begin{qbox}[title={M10.2: Tanda konstanta pemisah\hfill \level{L3}\;\src{Var}}]
\small Untuk $\lambda=k^2$, $\lambda=0$, dan $\lambda=-k^2$, tuliskan keluarga bentuk solusi $X$ dan $Y$. Jelaskan mengapa kondisi batas menentukan tanda yang berguna.
\end{qbox}
\begin{qbox}[title={M10.3: Boundary kiri-kanan nol\hfill \level{L3}\;\src{Var}}]
\small Pada domain $0<x<L$, syarat $X(0)=X(L)=0$ diterapkan pada $X''+k^2X=0$. Tunjukkan bahwa $k=n\pi/L$ dan $X_n=\sin(n\pi x/L)$.
\end{qbox}
\begin{qbox}[title={M10.4: Seleksi alternatif solusi\hfill \level{L3}\;\src{Var}}]
\small Diberikan empat kandidat berbasis $\sin,\cos,e^{ky},e^{-ky}$. Gunakan kondisi $T(0,y)=T(L,y)=0$ dan keterbatasan saat $y\to\infty$ untuk mengeliminasi kandidat yang tidak mungkin.
\end{qbox}
\begin{qbox}[title={M10.5: Superposisi mode\hfill \level{L3}\;\src{Var}}]
\small Mengapa satu mode $\sin(n\pi x/L)e^{-n\pi y/L}$ umumnya tidak cukup memenuhi kondisi batas $T(x,0)=f(x)$? Jelaskan kebutuhan penjumlahan tak hingga.
\end{qbox}
\begin{qbox}[title={M10.6: Separation pada wave equation\hfill \level{L3}\;\src{Var}}]
\small Untuk $u_{tt}=c^2u_{xx}$, ambil $u=X(x)G(t)$. Turunkan dua ODE terpisah dan interpretasikan konstanta pemisah.
\end{qbox}
\begin{qbox}[title={M10.7: Fixed ends\hfill \level{L3}\;\src{Var}}]
\small Gunakan $u(0,t)=u(L,t)=0$ untuk menunjukkan bahwa mode ruang tali berupa $\sin(n\pi x/L)$.
\end{qbox}
\begin{qbox}[title={M10.8: Peta proses\hfill \level{L2}\;\src{Var}}]
\small Tuliskan urutan lengkap penyelesaian BVP analitik: model $\to$ separation $\to$ solusi ODE $\to$ kondisi batas $\to$ superposisi $\to$ Fourier $\to$ verifikasi.
\end{qbox}
\end{multicols}

\end{qbox}
\begin{qbox}[title={M12.7: Orthogonality extraction\hfill \level{L3}\;\src{Var}}]
\small Dari $f(x)=\sum B_n\sin(n\pi x/L)$, kalikan dengan $\sin(m\pi x/L)$ dan integralkan pada $[0,L]$ untuk memperoleh formula $B_m$.
\end{qbox}
\begin{qbox}[title={M12.8: Uji batas bawah\hfill \level{L3}\;\src{Var}}]
\small Ambil solusi seri hasil M12.5 dan evaluasi numerik di $y=0$ menggunakan 5, 20, 100 mode. Jelaskan konvergensi dekat titik diskontinu $x=5$.
\end{qbox}
\end{multicols}

\section*{M14: Persamaan Gelombang dan Vibrating String}
{\color{iteraGray}\small Mode normal, fixed ends, dan kondisi awal.}\par\vspace{1mm}
\begin{multicols}{2}
\begin{qbox}[title={M14.1: Wave equation 1D\hfill \level{L2}\;\src{Var}}]
\small Tuliskan arti setiap variabel dan parameter pada $u_{tt}=c^2u_{xx}$. Verifikasikan bahwa $u=A\cos(kx-\omega t)$ memenuhi persamaan jika $\omega=ck$.
\end{qbox}
\begin{qbox}[title={M14.2: Mode fixed-end\hfill \level{L3}\;\src{Var}}]
\small Dengan $u(0,t)=u(L,t)=0$, turunkan mode ruang $X_n=\sin(n\pi x/L)$ dan frekuensi alami $\omega_n=n\pi c/L$.
\end{qbox}
\begin{qbox}[title={M14.3: Solusi umum modal\hfill \level{L3}\;\src{Var}}]
\small Tuliskan bentuk $u(x,t)=\sum [A_n\cos(\omega_nt)+B_n\sin(\omega_nt)]\sin(n\pi x/L)$. Jelaskan arti $A_n$ dan $B_n$.
\end{qbox}
\begin{qbox}[title={M14.4: Kondisi awal ke koefisien\hfill \level{L3}\;\src{Var}}]
\small Jika $u(x,0)=f(x)$ dan $u_t(x,0)=0$, tunjukkan bahwa $B_n=0$ dan $A_n$ diperoleh dari sine-series $f(x)$.
\end{qbox}
\begin{qbox}[title={M14.5: Tali segitiga\hfill \level{L4}\;\src{PR}}]
\small Untuk simpangan awal segitiga dengan puncak $h$ di $x=L/2$ dan kedua ujung nol, turunkan koefisien mode $A_n$. Tunjukkan mengapa hanya mode ganjil tertentu yang dominan/nonzero.
\end{qbox}
\begin{qbox}[title={M14.6: Energi modal secara kualitatif\hfill \level{L3}\;\src{Var}}]
\small Jika bentuk awal lebih tajam, bagaimana Anda menduga distribusi koefisien mode tinggi dibanding bentuk awal yang halus? Jelaskan secara Fourier.
\end{qbox}
\end{multicols}

\section*{Mock UAS}
\begin{exam}[title={Mock UAS A - Pelat dengan batas bawah piecewise}]
Gunakan $XY$ = dua digit terakhir NIM. Sebuah pelat memenuhi $T_{xx}+T_{yy}=0$ pada $0<x<1$, $0<y<2$. Kondisi batas: $T(0,y)=T(1,y)=T(x,2)=0$, sedangkan
\[
T(x,0)=\begin{cases}XY/2,&0<x<0.5,\\XY,&0.5<x<1.\end{cases}
\]
\textbf{Tugas:} (1) lakukan separation of variables; (2) tentukan mode yang memenuhi tiga batas homogen; (3) hitung koefisien Fourier dari batas bawah; (4) tulis solusi seri $T(x,y)$; (5) jelaskan cara memverifikasi kondisi batas dan PDE; (6) buat sketsa/plot kontur untuk satu nilai $XY$ pilihan.
\end{exam}

\begin{exam}[title={Mock UAS B - Pelat semi-infinite dengan variasi domain}]
Pada domain $0<x<L$, $y>0$, diketahui $T_{xx}+T_{yy}=0$, $T(0,y)=T(L,y)=0$, $T\to0$ saat $y\to\infty$, dan
\[
T(x,0)=T_0\left(1-\frac{x}{L}\right),\qquad 0<x<L.
\]
Tentukan solusi analitik dalam bentuk seri. Hitung formula koefisien secara eksplisit, lalu evaluasi pendekatan 5 dan 20 mode pada $(x,y)=(L/2,L/4)$.
\end{exam}

\begin{exam}[title={Mock UAS C - Integrasi BVP dan Fourier}]
Pelat $0<x<L$, $0<y<H$ memenuhi persamaan Laplace. Tiga sisi $x=0$, $x=L$, dan $y=H$ bernilai nol. Batas bawah berupa fungsi segitiga:
\[
f(x)=\begin{cases}
2T_0x/L,&0\le x\le L/2,\\
2T_0(1-x/L),&L/2<x\le L.
\end{cases}
\]
Bangun solusi dengan separation of variables dan deret Fourier. Identifikasi parity relatif terhadap titik tengah, turunkan koefisien, dan jelaskan pengaruh $H$ terhadap faktor pada arah $y$.
\end{exam}

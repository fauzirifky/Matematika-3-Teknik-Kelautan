\begin{frame}{Mengapa Fourier Muncul di Teknik Kelautan?}
Banyak besaran berubah berulang terhadap waktu atau posisi:
\begin{itemize}
\item elevasi gelombang dan pasang surut;
\item gerak heave, pitch, dan roll;
\item beban periodik pada struktur;
\item getaran mesin dan sinyal sensor.
\end{itemize}
\begin{block}{Ide Fourier}
Bentuk periodik yang rumit dapat diuraikan menjadi gabungan sinus dan cosinus dengan frekuensi berbeda.
\end{block}
\[
\text{sinyal periodik}\longleftrightarrow\text{komponen harmonik}
\]
\end{frame}


\begin{frame}{Fungsi Periodik}
\begin{block}{Definisi}
$f$ periodik jika ada $T>0$ sehingga
\[
f(x+T)=f(x).
\]
Nilai positif terkecil yang memenuhi disebut periode dasar.
\end{block}
Untuk fungsi berperiode $2\pi$, cukup definisikan $f(x)$ pada $(-\pi,\pi)$ lalu lanjutkan secara periodik.
\end{frame}


\begin{frame}{Bentuk Umum Deret Fourier}
Untuk periode $2\pi$:
\[
\boxed{f(x)=\frac{a_0}{2}+\sum_{n=1}^{\infty}\left[a_n\cos(nx)+b_n\sin(nx)\right]}
\]
\begin{itemize}
\item $a_0/2$: komponen rata-rata;
\item $a_n$: bobot harmonik cosinus ke-$n$;
\item $b_n$: bobot harmonik sinus ke-$n$.
\end{itemize}
Pertanyaan kita sekarang: dari mana koefisien-koefisien itu datang?
\end{frame}


\begin{frame}{Ortogonalitas Sinus dan Cosinus}
Untuk $m,n=1,2,\ldots$:
\[
\int_{-\pi}^{\pi}\cos(mx)\cos(nx)\,dx=
\begin{cases}0,&m\ne n,\\ \pi,&m=n,\end{cases}
\]
\[
\int_{-\pi}^{\pi}\sin(mx)\sin(nx)\,dx=
\begin{cases}0,&m\ne n,\\ \pi,&m=n,\end{cases}
\]
dan
\[
\int_{-\pi}^{\pi}\sin(mx)\cos(nx)\,dx=0.
\]
\begin{block}{Makna}
Kalikan deret dengan satu harmonik lalu integralkan: harmonik lain lenyap.
\end{block}
\end{frame}


\begin{frame}{Menurunkan Koefisien Fourier}
Integralkan deret pada $[-\pi,\pi]$:
\[
\boxed{a_0=\frac1\pi\int_{-\pi}^{\pi}f(x)\,dx.}
\]
Kalikan deret dengan $\cos(mx)$ lalu integralkan:
\[
\boxed{a_n=\frac1\pi\int_{-\pi}^{\pi}f(x)\cos(nx)\,dx.}
\]
Kalikan deret dengan $\sin(mx)$ lalu integralkan:
\[
\boxed{b_n=\frac1\pi\int_{-\pi}^{\pi}f(x)\sin(nx)\,dx.}
\]
\end{frame}


\begin{frame}{Jika Periodenya $2L$}
\[
f(x)=\frac{a_0}{2}+\sum_{n=1}^{\infty}\left[
a_n\cos\left(\frac{n\pi x}{L}\right)+
b_n\sin\left(\frac{n\pi x}{L}\right)\right],
\]
\[
a_0=\frac1L\int_{-L}^{L}f(x)\,dx,
\]
\[
a_n=\frac1L\int_{-L}^{L}f(x)\cos\left(\frac{n\pi x}{L}\right)dx,\quad
b_n=\frac1L\int_{-L}^{L}f(x)\sin\left(\frac{n\pi x}{L}\right)dx.
\]
\begin{alertblock}{Perhatikan}
$\cos(nx)$ dan $\sin(nx)$ adalah bentuk khusus ketika $L=\pi$.
\end{alertblock}
\end{frame}


\begin{frame}{Genap dan Ganjil: Jalan Pintas}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Genap: $f(-x)=f(x)$}
\[
b_n=0,\qquad
a_n=\frac2\pi\int_0^\pi f(x)\cos(nx)\,dx.
\]
Deret hanya memuat cosinus.
\end{block}
\column{.48\textwidth}
\begin{block}{Ganjil: $f(-x)=-f(x)$}
\[
a_0=a_n=0,\qquad
b_n=\frac2\pi\int_0^\pi f(x)\sin(nx)\,dx.
\]
Deret hanya memuat sinus.
\end{block}
\end{columns}
\end{frame}


\begin{frame}{Contoh 1 - Pulsa Kotak Periodik}
Untuk satu periode:
\[
f(x)=
\begin{cases}
0,&-\pi<x<0,\\
A,&0<x<\pi,
\end{cases}
\qquad f(x+2\pi)=f(x).
\]
Tentukan deret Fourier-nya.
\pauseboard
\end{frame}


\begin{frame}{Contoh 1 - Menyusun Integral}
\[
a_0=\frac1\pi\left[\int_{-\pi}^{0}0\,dx+\int_0^\pi A\,dx\right],
\]
\[
a_n=\frac1\pi\int_0^\pi A\cos(nx)\,dx,\qquad
b_n=\frac1\pi\int_0^\pi A\sin(nx)\,dx.
\]
\pauseboard
\end{frame}


\begin{frame}{Contoh 1 - Jawaban}
\[
a_0=A,\qquad a_n=0,
\]
\[
b_n=\frac{A}{n\pi}\left(1-(-1)^n\right)
=
\begin{cases}
\dfrac{2A}{n\pi},&n\text{ ganjil},\\
0,&n\text{ genap}.
\end{cases}
\]
Maka
\[
\boxed{
f(x)=\frac A2+\frac{2A}{\pi}
\left(\sin x+\frac13\sin3x+\frac15\sin5x+\cdots\right).
}
\]
\end{frame}


\begin{frame}{Membaca Hasil Pulsa Kotak}
\begin{itemize}
\item nilai rata-rata fungsi adalah $A/2$;
\item hanya harmonik sinus ganjil yang muncul;
\item amplitudo harmonik menurun seperti $1/n$;
\item makin banyak harmonik, makin dekat rekonstruksi ke pulsa kotak.
\end{itemize}
\begin{block}{Di titik loncatan}
Deret konvergen ke rata-rata limit kiri dan kanan. Di sekitar loncatan muncul fenomena Gibbs.
\end{block}
\end{frame}


\begin{frame}{Contoh 2 - Fungsi Piecewise Linier}
Diberikan fungsi berperiode $2\pi$:
\[
g(x)=
\begin{cases}
0,&-\pi<x<0,\\
\dfrac{2x}{\pi},&0<x<\pi.
\end{cases}
\]
Susun integral untuk $a_0,a_n,b_n$.
\pauseboard
\end{frame}


\begin{frame}{Contoh 2 - Jawaban}
\[
a_0=\frac1\pi\int_0^\pi\frac{2x}{\pi}\,dx=1,
\]
\[
a_n=\frac1\pi\int_0^\pi\frac{2x}{\pi}\cos(nx)\,dx
=\frac{2\left((-1)^n-1\right)}{\pi^2n^2},
\]
\[
b_n=\frac1\pi\int_0^\pi\frac{2x}{\pi}\sin(nx)\,dx
=-\frac{2(-1)^n}{\pi n}.
\]
Jadi
\[
g(x)=\frac12+\sum_{n=1}^{\infty}\left[
\frac{2\left((-1)^n-1\right)}{\pi^2n^2}\cos(nx)
-\frac{2(-1)^n}{\pi n}\sin(nx)\right].
\]
\end{frame}


\begin{frame}{Pergeseran Fungsi dan Fase}
Jika $h(x)=f(x+\alpha)$, grafik bergeser ke kiri sejauh $\alpha$.
\[
\cos(n(x+\alpha))=\cos(nx)\cos(n\alpha)-\sin(nx)\sin(n\alpha),
\]
\[
\sin(n(x+\alpha))=\sin(nx)\cos(n\alpha)+\cos(nx)\sin(n\alpha).
\]
\begin{block}{Akibat}
Pergeseran tidak menambah frekuensi baru; yang berubah adalah fase dan pasangan koefisien sinus-cosinus.
\end{block}
\end{frame}


\begin{frame}{Contoh 3 - Pergeseran}
Diketahui
\[
f(x)=2+3\cos x-4\sin x.
\]
Tuliskan $h(x)=f(x+\pi/2)$ dalam bentuk
\[
\frac{a_0}{2}+a_1\cos x+b_1\sin x.
\]
\pauseboard
\end{frame}


\begin{frame}{Contoh 3 - Jawaban}
\[
h(x)=2+3\cos\left(x+\frac\pi2\right)
-4\sin\left(x+\frac\pi2\right).
\]
Karena
\[
\cos(x+\pi/2)=-\sin x,\qquad
\sin(x+\pi/2)=\cos x,
\]
maka
\[
\boxed{h(x)=2-4\cos x-3\sin x.}
\]
Frekuensi tetap, tetapi koefisien berubah akibat fase.
\end{frame}


\begin{frame}{Menjawab Pertanyaan Aplikasi secara Utuh}
Jangan berhenti pada kalimat sederhana bahwa Fourier hanya digunakan untuk menganalisis gelombang.
\begin{block}{Struktur}
\textbf{Fenomena} $\rightarrow$ \textbf{uraian harmonik} $\rightarrow$ \textbf{informasi} $\rightarrow$ \textbf{kegunaan teknik}.
\end{block}
\begin{itemize}
\item elevasi muka air $\rightarrow$ harmonik $\rightarrow$ frekuensi/amplitudo dominan $\rightarrow$ analisis gelombang atau pasut;
\item respons struktur $\rightarrow$ harmonik respons $\rightarrow$ frekuensi dominan $\rightarrow$ identifikasi resonansi/beban;
\item sinyal sensor $\rightarrow$ spektrum $\rightarrow$ noise dan komponen utama $\rightarrow$ pemrosesan pengukuran.
\end{itemize}
\end{frame}


\begin{frame}{Deret Fourier dan Transformasi Fourier}
\begin{columns}[T]
\column{.48\textwidth}
\begin{block}{Deret Fourier}
Untuk fungsi periodik; frekuensi muncul sebagai harmonik diskrit:
\[
n\omega_0,\quad n=0,1,2,\ldots
\]
\end{block}
\column{.48\textwidth}
\begin{block}{Transformasi Fourier}
Lebih umum, termasuk untuk sinyal nonperiodik; spektrum dapat kontinu.
\end{block}
\end{columns}
\vspace{2mm}
Pertemuan ini menuntaskan \textbf{Deret Fourier}. Judul Transformasi Fourier I mengikuti kontrak perkuliahan.
\end{frame}

\begin{frame}[fragile]{MATLAB Online - Skrip Siap Run (1/2)}
\begin{lstlisting}[style=matlab]
%% Fourier series: periodic square pulse
clear; clc; close all;


A = 1;
N = 15;
x = linspace(-3*pi,3*pi,5000);


% f=0 on (-pi,0), f=A on (0,pi), period 2*pi
phase = mod(x,2*pi);
f = A*(phase > 0 & phase < pi);


S = (A/2)*ones(size(x));
for n = 1:N
    an = 0;
    bn = A/(pi*n)*(1-(-1)^n);
    S = S + an*cos(n*x) + bn*sin(n*x);
end
\end{lstlisting}
\end{frame}


\begin{frame}[fragile]{MATLAB Online - Skrip Siap Run (2/2)}
\begin{lstlisting}[style=matlab]
figure;
plot(x,f,'LineWidth',1.6); hold on;
plot(x,S,'LineWidth',1.4);
grid on; xlim([-2*pi 2*pi]); ylim([-0.3 1.3]);
xticks([-2*pi -pi 0 pi 2*pi]);
xticklabels({'-2\pi','-\pi','0','\pi','2\pi'});
xlabel('x'); ylabel('f(x)');
legend('Original','Fourier S_N','Location','best');
title(sprintf('Square pulse Fourier series, N = %d',N));


fprintf(' n        a_n             b_n\n');
for n = 1:min(N,12)
    an = 0;
    bn = A/(pi*n)*(1-(-1)^n);
    fprintf('%2d   %12.6f   %12.6f\n',n,an,bn);
end
\end{lstlisting}
\end{frame}

\begin{frame}{Latihan Penutup}

\[
q(x)=
\begin{cases}
-2,&-\pi<x<0,\\
2,&0<x<\pi,
\end{cases}
\]
tentukan:
\begin{enumerate}
\item apakah $q$ genap atau ganjil;
\item koefisien mana yang otomatis nol;
\item integral koefisien yang masih perlu dihitung;
\item tiga harmonik tak nol pertama.
\end{enumerate}

\begin{frame}{Ringkasan}
\[
f(x)=\frac{a_0}{2}+\sum_{n=1}^{\infty}
[a_n\cos(nx)+b_n\sin(nx)].
\]
\begin{enumerate}
\item Koefisien muncul dari ortogonalitas sinus dan cosinus.
\item Fungsi piecewise dihitung dengan memisahkan integral sesuai intervalnya.
\item Simetri dapat memangkas banyak perhitungan.
\item Pergeseran mengubah fase, bukan daftar frekuensi harmonik.
\item Jumlah parsial memberi rekonstruksi numerik; diskontinuitas menimbulkan Gibbs.
\item Fourier memberi bahasa untuk membaca kandungan harmonik fenomena periodik teknik kelautan.
\end{enumerate}
\end{frame}

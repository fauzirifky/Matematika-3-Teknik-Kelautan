\begin{frame}{Pemanasan 1: Transformasi Langsung}
Tanpa melihat catatan, tentukan:
\begin{enumerate}
\item $\Lap{5t^2}$
\item $\Lap{3\sin 3t}$
\item $\Lap{4-2\cos 2t}$
\item $\Lap{e^{-2t}}$
\item $\Lap{2t^3+5\cos 4t}$
\end{enumerate}
\vspace{2mm}
\begin{exampleblock}{Strategi}
Pisahkan dengan linearitas, lalu cocokkan satu per satu dengan tabel.
\end{exampleblock}

\begin{frame}{Mengapa Transformasi Turunan Penting?}
PDB hampir selalu berisi $y'$, $y''$, dan seterusnya.
\[
y''+4y'+4y=f(t).
\]
Jika hanya tahu $\Lap{y}$, kita belum bisa mentransformasikan seluruh persamaan.
\vspace{2mm}
\begin{block}{Pertanyaan utama}
Jika $Y(s)=\Lap{y(t)}$, apa bentuk $\Lap{y'(t)}$ dan $\Lap{y''(t)}$?
\end{block}
\end{frame}

\begin{frame}{Menurunkan $\Lap{y'}$: Mulai dari Definisi}
\[
\Lap{y'}=\int_0^\infty e^{-st}y'(t)\,dt.
\]
Gunakan integral parsial:
\[
\int u\,dv=uv-\int v\,du.
\]
Ambil
\[
u=e^{-st},\qquad dv=y'(t)dt.
\]
Maka
\[
du=-se^{-st}dt,\qquad v=y(t).
\]
\end{frame}

\begin{frame}{Hasil $\Lap{y'}$}
\begin{align*}
\Lap{y'}
&=\left[e^{-st}y(t)\right]_0^\infty
+s\int_0^\infty e^{-st}y(t)\,dt\\
&=-y(0)+sY(s).
\end{align*}
\begin{alertblock}{Rumus penting}
\[
\boxed{\Lap{y'}=sY(s)-y(0)}
\]
\end{alertblock}
\begin{itemize}
\item Turunan berubah menjadi perkalian oleh $s$.
\item Kondisi awal $y(0)$ otomatis muncul.
\end{itemize}
\end{frame}

\begin{frame}{Dari $y'$ ke $y''$}
Anggap $y''=(y')'$. Terapkan rumus yang sama:
\[
\Lap{y''}=s\Lap{y'}-y'(0).
\]
Substitusikan $\Lap{y'}=sY-y(0)$:
\[
\Lap{y''}=s[sY-y(0)]-y'(0).
\]
\begin{alertblock}{Rumus penting}
\[
\boxed{\Lap{y''}=s^2Y-sy(0)-y'(0)}
\]
\end{alertblock}
\end{frame}

\begin{frame}{Pola Turunan Orde Tinggi}
\small
\[
\Lap{y^{(n)}}=s^nY-s^{n-1}y(0)-s^{n-2}y'(0)-\cdots-y^{(n-1)}(0).
\]
Contoh orde tiga:
\[
\Lap{y'''}=s^3Y-s^2y(0)-sy'(0)-y''(0).
\]
\begin{block}{Cara mengingat}
Pangkat $s$ turun satu per satu, sedangkan orde kondisi awal naik satu per satu.
\end{block}
\vfill
\boasnote
\end{frame}

\begin{frame}{Latihan Kelas 1: Transformasi Turunan}
Diketahui $Y(s)=\Lap{y(t)}$. Tuliskan transformasi dari:
\begin{enumerate}
\item $y'$ jika $y(0)=4$;
\item $y''$ jika $y(0)=2$ dan $y'(0)=-1$;
\item $y'''$ jika $y(0)=0$, $y'(0)=3$, $y''(0)=5$;
\item $2y''-3y'+7y$ dengan $y(0)=1$, $y'(0)=2$.
\end{enumerate}
\end{frame}

\begin{frame}{Contoh: Transformasikan PDB, Belum Perlu Diselesaikan}
Diberikan
\[
y''+3y'+2y=4t,\qquad y(0)=1,\quad y'(0)=0.
\]
Transformasikan setiap suku:
\[
(s^2Y-s)+(3sY-3)+2Y=\frac{4}{s^2}.
\]
Gabungkan:
\[
(s^2+3s+2)Y-s-3=\frac{4}{s^2}.
\]
\begin{exampleblock}{Yang terjadi}
PDB di $t$ berubah menjadi \textbf{persamaan aljabar dalam $Y(s)$}.
\end{exampleblock}
\end{frame}

\begin{frame}{Invers Laplace: Jalan Pulang}
Jika
\[
F(s)=\Lap{f(t)},
\]
maka
\[
f(t)=\invLap{F(s)}.
\]
\begin{block}{Ide}
Kita tidak biasanya menghitung integral invers dari nol. Kita merapikan $F(s)$ agar cocok dengan tabel transformasi.
\end{block}
\centering
\Large
$F(s)\ \xrightarrow{\text{rapikan}}\ \text{bentuk tabel}\ \xrightarrow{\Lcal^{-1}}\ f(t)$
\end{frame}

\begin{frame}{Contoh Invers Laplace Dasar}
\begin{exampleblock}{Contoh 1}
\[
\invLap{\frac{6}{s^4}}
=6\invLap{\frac{1}{s^4}}
=6\frac{t^3}{3!}=t^3.
\]
\end{exampleblock}
\begin{exampleblock}{Contoh 2}
\[
\invLap{\frac{5}{s^2+25}}=\sin 5t.
\]
\end{exampleblock}
\begin{exampleblock}{Contoh 3}
\[
\invLap{\frac{s}{s^2+9}}=\cos 3t.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Latihan Kelas 2: Invers dari Tabel}
Tentukan:
\begin{enumerate}
\item $\displaystyle \invLap{\frac{1}{s+4}}$
\item $\displaystyle \invLap{\frac{12}{s^3}}$
\item $\displaystyle \invLap{\frac{7}{s^2+49}}$
\item $\displaystyle \invLap{\frac{s}{s^2+16}}$
\item $\displaystyle \invLap{\frac{3s+8}{s^2+4}}$
\end{enumerate}
\end{frame}

\begin{frame}{Masalah Nyata: Bentuknya Tidak Selalu Ada di Tabel}
Misalnya
\[
Y(s)=\frac{3s+5}{s(s+2)}.
\]
Tidak ada satu baris tabel yang langsung cocok.
\vspace{2mm}
\begin{alertblock}{Solusi}
Pecah menjadi beberapa pecahan sederhana:
\[
\frac{3s+5}{s(s+2)}=\frac{A}{s}+\frac{B}{s+2}.
\]
Inilah \textbf{pecahan parsial}.
\end{alertblock}
\end{frame}

\begin{frame}{Pecahan Parsial: Faktor Linear Berbeda}
Untuk
\[
\frac{P(s)}{(s-a)(s-b)},
\]
ambil
\[
\frac{P(s)}{(s-a)(s-b)}=\frac{A}{s-a}+\frac{B}{s-b}.
\]
\begin{exampleblock}{Contoh}
\[
\frac{3s+5}{s(s+2)}=\frac{A}{s}+\frac{B}{s+2}.
\]
Kalikan $s(s+2)$:
\[
3s+5=A(s+2)+Bs.
\]
Set $s=0\Rightarrow A=\frac52$; set $s=-2\Rightarrow B=\frac12$.
\end{exampleblock}
\end{frame}

\begin{frame}{Dari Pecahan Parsial ke Invers}
Dari
\[
Y(s)=\frac{5/2}{s}+\frac{1/2}{s+2},
\]
maka
\[
y(t)=\frac52+\frac12e^{-2t}.
\]
\begin{block}{Pola kerja}
\begin{enumerate}
\item faktorkan penyebut;
\item pecah ke pecahan parsial;
\item cocokkan setiap suku dengan tabel;
\item ambil invers Laplace.
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}{Pecahan Parsial: Faktor Berulang}
Jika penyebut mengandung $(s+a)^2$, jangan hanya tulis satu suku.
\[
\frac{P(s)}{s(s+2)^2}
=\frac{A}{s}+\frac{B}{s+2}+\frac{C}{(s+2)^2}.
\]
\begin{alertblock}{Ingat}
Untuk faktor berulang $(s+a)^m$, tuliskan semua pangkat:
\[
\frac{A_1}{s+a}+\frac{A_2}{(s+a)^2}+\cdots+\frac{A_m}{(s+a)^m}.
\]
\end{alertblock}
\end{frame}

\begin{frame}{Penyebut Kuadrat: Lengkapi Kuadrat}
Bentuk
\[
s^2+4s+13
\]
dirubah menjadi
\[
(s+2)^2+9.
\]
Kemudian pecah pembilang menjadi bagian $(s+2)$ dan konstanta.
\begin{exampleblock}{Contoh}
\[
\frac{-s+1}{s^2+4s+13}
= -\frac{s+2}{(s+2)^2+9}+\frac{3}{(s+2)^2+9}.
\]
Masing-masing cocok dengan bentuk eksponensial-kosinus dan eksponensial-sinus.
\end{exampleblock}
\end{frame}

\begin{frame}{Contoh Boas: Mengapa Pecahan Parsial Muncul}
Pada salah satu contoh Boas,
\[
Y(s)=\frac{s^2+8s+27}{(s+1)(s^2+4s+13)}.
\]
Setelah pecahan parsial dan melengkapi kuadrat:
\[
Y(s)=\frac{2}{s+1}
-\frac{s+2}{(s+2)^2+9}
+\frac{3}{(s+2)^2+9}.
\]
Sehingga bentuk waktunya merupakan kombinasi
\[
e^{-t},\qquad e^{-2t}\cos 3t,\qquad e^{-2t}\sin 3t.
\]
\vfill
\boasnote
\end{frame}

\begin{frame}{Latihan Kelas 3: Pecahan Parsial}
Uraikan, lalu tentukan invers Laplace bila memungkinkan:
\begin{enumerate}
\item $\displaystyle \frac{2s+3}{s(s+1)}$
\item $\displaystyle \frac{s+5}{(s+1)(s+3)}$
\item $\displaystyle \frac{1}{s(s+2)^2}$
\item $\displaystyle \frac{2s+7}{s^2+4s+13}$
\item $\displaystyle \frac{3s+10}{s^2-25}$
\end{enumerate}
\end{frame}

\begin{frame}{Shifting I: Pergeseran di Domain $s$}
Jika
\[
\Lap{f(t)}=F(s),
\]
maka
\[
\boxed{\Lap{e^{at}f(t)}=F(s-a)}.
\]
Atau ekuivalen:
\[
\boxed{\Lap{e^{-at}f(t)}=F(s+a)}.
\]
\begin{exampleblock}{Contoh}
Karena $\Lap{\sin bt}=\dfrac{b}{s^2+b^2}$,
\[
\Lap{e^{-at}\sin bt}
=\frac{b}{(s+a)^2+b^2}.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Latihan Kelas 4: Shifting I}
Tentukan transformasi atau inversnya:
\begin{enumerate}
\item $\Lap{e^{-3t}\cos 2t}$
\item $\Lap{e^{2t}\sin 5t}$
\item $\displaystyle \invLap{\frac{4}{(s+1)^2+16}}$
\item $\displaystyle \invLap{\frac{s-3}{(s-3)^2+9}}$
\item $\displaystyle \invLap{\frac{6}{(s+2)^2+36}}$
\end{enumerate}
\end{frame}

\begin{frame}{Shifting II: Fungsi yang Mulai Terlambat}
Definisikan fungsi step Heaviside
\[
u(t-a)=\begin{cases}
0,& t<a,\\
1,& t\ge a.
\end{cases}
\]
Jika $G(s)=\Lap{g(t)}$, maka
\[
\boxed{\Lap{g(t-a)u(t-a)}=e^{-as}G(s)}.
\]
\begin{block}{Makna}
Faktor $e^{-as}$ di domain Laplace menandakan pergeseran waktu sebesar $a$.
\end{block}
\end{frame}

\begin{frame}{Contoh Pergeseran Waktu}
Tentukan invers:
\[
F(s)=e^{-\pi s}\frac{s}{s^2+1}.
\]
Tanpa $e^{-\pi s}$:
\[
\invLap{\frac{s}{s^2+1}}=\cos t.
\]
Dengan shifting:
\[
\boxed{f(t)=\cos(t-\pi)\,u(t-\pi)}.
\]
\begin{exampleblock}{Baca secara fisik}
Sinyal kosinus ``dinyalakan'' mulai $t=\pi$.
\end{exampleblock}
\end{frame}

\begin{frame}{Latihan Kelas 5: Shifting II}
Tentukan invers Laplace:
\begin{enumerate}
\item $\displaystyle e^{-2s}\frac{1}{s}$
\item $\displaystyle e^{-3s}\frac{2}{s^2+4}$
\item $\displaystyle e^{-s}\frac{s}{s^2+9}$
\end{enumerate}
Kemudian gambarkan secara skematik fungsi waktunya: kapan fungsi bernilai nol, dan kapan mulai aktif?
\end{frame}

\begin{frame}{Jembatan ke Penyelesaian PDB}
Pertemuan berikutnya akan memakai alur berikut:
\begin{enumerate}
\item transformasikan semua suku PDB;
\item masukkan kondisi awal;
\item selesaikan persamaan aljabar untuk $Y(s)$;
\item pecah/rapikan $Y(s)$;
\item ambil invers Laplace untuk mendapatkan $y(t)$;
\item cek kondisi awal dan, bila perlu, substitusikan kembali.
\end{enumerate}
\end{frame}

\begin{frame}{Contoh Mini IVP: Sampai Selesai}
Selesaikan
\[
y'+2y=1,\qquad y(0)=0.
\]
Transformasi:
\[
sY-0+2Y=\frac1s.
\]
Maka
\[
Y=\frac{1}{s(s+2)}
=\frac12\frac1s-\frac12\frac1{s+2}.
\]
Invers:
\[
\boxed{y(t)=\frac12\left(1-e^{-2t}\right)}.
\]
\end{frame}

\begin{frame}{Latihan Kelas 6: IVP Ringan}
Selesaikan dengan Transformasi Laplace:
\begin{enumerate}
\item $y'+y=2$, $y(0)=1$.
\item $y'+3y=e^{-t}$, $y(0)=0$.
\item $y''+4y=0$, $y(0)=2$, $y'(0)=0$.
\item $y''+y=\sin t$, $y(0)=1$, $y'(0)=0$.
\end{enumerate}
\begin{alertblock}{Untuk nomor 4}
Jika bentuk aljabarnya terasa tidak langsung, justru itu latihan untuk merapikan $Y(s)$ sebelum invers.
\end{alertblock}
\end{frame}

\begin{frame}{Latihan Kelas 7: Pilih Strategi}
Untuk setiap bentuk berikut, sebutkan \textbf{strategi pertama} yang harus dilakukan:
\begin{enumerate}
\item $\displaystyle \frac{4}{s^2+16}$
\item $\displaystyle \frac{s+2}{(s+2)^2+9}$
\item $\displaystyle \frac{3s+5}{s(s+2)}$
\item $\displaystyle e^{-2s}\frac{1}{s^2+1}$
\item $\displaystyle \frac{2s+7}{s^2+4s+13}$
\end{enumerate}
Pilihan strategi: \emph{langsung tabel, shifting, pecahan parsial, melengkapi kuadrat}.
\end{frame}

\begin{frame}{Ringkasan Rumus Hari Ini}
\small
\begin{align*}
\Lap{y'} &= sY-y(0),\\
\Lap{y''} &= s^2Y-sy(0)-y'(0),\\
\Lap{y^{(n)}} &= s^nY-s^{n-1}y(0)-\cdots-y^{(n-1)}(0),\\
\Lap{e^{-at}f(t)} &= F(s+a),\\
\Lap{f(t-a)u(t-a)} &= e^{-as}F(s).
\end{align*}
\begin{alertblock}{Skill inti}
Rumus penting, tetapi kemampuan paling penting hari ini adalah \textbf{merapikan bentuk aljabar agar cocok dengan tabel invers}.
\end{alertblock}
\end{frame}

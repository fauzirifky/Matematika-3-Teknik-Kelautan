\begin{frame}{Posisi Pertemuan 4}
\begin{block}{Pekan 2--4: Transformasi Laplace}
Pertemuan ini menutup blok Transformasi Laplace sebelum masuk blok berikutnya. Fokusnya adalah latihan yang paling sering muncul pada soal ujian: konsep, transformasi fungsi, transformasi balik, dan penyelesaian PDB.
\end{block}
\begin{exampleblock}{Target praktis}
Mahasiswa mampu mengubah masalah dari domain waktu ke domain $s$, menyelesaikan bentuk aljabar yang muncul, lalu kembali ke fungsi waktu.
\end{exampleblock}
\[
f(t)\quad \xrightarrow{\;\mathcal L\;}\quad F(s)\quad \xrightarrow{\text{aljabar}}\quad Y(s)\quad \xrightarrow{\;\mathcal L^{-1}\;}\quad y(t)
\]
\end{frame}

\begin{frame}{Orientasi Latihan UTS}
\begin{enumerate}
\item Menjelaskan definisi dan kegunaan Transformasi Laplace.
\item Menghitung transformasi Laplace fungsi dasar dan gabungannya.
\item Menghitung transformasi balik dengan membaca tabel dan pecahan parsial.
\item Menyelesaikan PDB dengan kondisi awal menggunakan Transformasi Laplace.
\end{enumerate}
\begin{alertblock}{Pegangan pengerjaan}
Mulai dari definisi, cek tabel, rapikan bentuk aljabar, lalu tulis jawaban akhir dalam fungsi waktu.
\end{alertblock}
\end{frame}

\begin{frame}{Definisi yang Dipakai}
\begin{block}{Transformasi Laplace}
Untuk fungsi $f(t)$ dengan $t\ge 0$,
\[
F(s)=\Lap{f(t)}=\int_0^\infty e^{-st}f(t)\,dt.
\]
\end{block}
\begin{itemize}
\item Fungsi awal: $f(t)$, berada di domain waktu.
\item Hasil transformasi: $F(s)$, berada di domain $s$.
\item Jika integralnya tidak konvergen, transformasi Laplace biasa tidak ada.
\end{itemize}
\end{frame}

\begin{frame}{Contoh dari Definisi: $f(t)=1$}
\[
\Lap{1}=\int_0^\infty e^{-st}\,dt.
\]
Untuk $s>0$,
\[
\int_0^\infty e^{-st}\,dt
=\left[-\frac{1}{s}e^{-st}\right]_0^\infty
=0-\left(-\frac{1}{s}\right)=\frac{1}{s}.
\]
\begin{exampleblock}{Hasil}
\[
\boxed{\Lap{1}=\frac{1}{s}},\qquad s>0.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Contoh dari Definisi: Eksponen}
Hitung $\Lap{e^{at}}$.
\[
\Lap{e^{at}}=\int_0^\infty e^{-st}e^{at}\,dt
=\int_0^\infty e^{-(s-a)t}\,dt.
\]
Jika $s>a$,
\[
\int_0^\infty e^{-(s-a)t}\,dt=\frac{1}{s-a}.
\]
\begin{exampleblock}{Hasil}
\[
\boxed{\Lap{e^{at}}=\frac{1}{s-a}},\qquad s>a.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Contoh dari Definisi: Sinus}
Dengan integral parsial dua kali, diperoleh
\[
I=\Lap{\sin at}=\int_0^\infty e^{-st}\sin(at)\,dt.
\]
Hasil akhirnya:
\[
\boxed{\Lap{\sin at}=\frac{a}{s^2+a^2}}.
\]
\begin{block}{Pasangannya}
\[
\boxed{\Lap{\cos at}=\frac{s}{s^2+a^2}}.
\]
\end{block}
\end{frame}

\begin{frame}{Dari Definisi ke Tabel}
Menghitung integral dari definisi selalu boleh, tetapi sering lambat. Karena itu kita memakai tabel.
\begin{center}\small
\begin{tabular}{c c}
\toprule
$f(t)$ & $\Lap{f(t)}$\\
\midrule
$1$ & $1/s$\\[1mm]
$t^n$ & $n!/s^{n+1}$\\[1mm]
$e^{at}$ & $1/(s-a)$\\[1mm]
$\sin at$ & $a/(s^2+a^2)$\\[1mm]
$\cos at$ & $s/(s^2+a^2)$\\
\bottomrule
\end{tabular}
\end{center}
\begin{alertblock}{Catatan}
Tabel lengkap L1--L35 ada pada PDF terpisah di folder tabel referensi.
\end{alertblock}
\end{frame}

\begin{frame}{Linearitas: Cara Menggabungkan Tabel}
\begin{block}{Sifat linear}
\[
\Lap{af(t)+bg(t)}=a\Lap{f(t)}+b\Lap{g(t)}.
\]
\end{block}
\begin{exampleblock}{Contoh}
\[
\Lap{2-5\sin(4t)}=2\Lap{1}-5\Lap{\sin(4t)}.
\]
\[
=\frac{2}{s}-5\frac{4}{s^2+16}
=\boxed{\frac{2}{s}-\frac{20}{s^2+16}}.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Eksponen Kali Trigonometri}
Gunakan pergeseran pada variabel $s$:
\[
\Lap{e^{at}g(t)}=G(s-a),\quad G(s)=\Lap{g(t)}.
\]
\begin{exampleblock}{Contoh 1}
\[
\Lap{e^{-t}\sin(3t)}=\frac{3}{(s+1)^2+9}.
\]
\end{exampleblock}
\begin{exampleblock}{Contoh 2}
\[
\Lap{e^{2t}\cos(4t)}=\frac{s-2}{(s-2)^2+16}.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Membuktikan Rumus dari Rumus Dasar}
Misal sudah diketahui
\[
\Lap{\sin bt}=\frac{b}{s^2+b^2}.
\]
Dengan sifat pergeseran,
\[
\Lap{e^{-at}\sin bt}
=\left.\frac{b}{s^2+b^2}\right|_{s\mapsto s+a}
=\boxed{\frac{b}{(s+a)^2+b^2}}.
\]
\begin{block}{Ide pembuktian}
Kalikan fungsi awal dengan eksponen, lalu geser variabel $s$ pada hasil transformasinya.
\end{block}
\end{frame}

\begin{frame}{Membuktikan Rumus dengan Turunan terhadap $s$}
Dari definisi,
\[
F(s)=\int_0^\infty e^{-st}f(t)\,dt.
\]
Turunkan terhadap $s$:
\[
F'(s)=\int_0^\infty (-t)e^{-st}f(t)\,dt=-\Lap{tf(t)}.
\]
Jadi
\[
\boxed{\Lap{tf(t)}=-\frac{d}{ds}F(s)}.
\]
\begin{exampleblock}{Contoh}
Jika $\Lap{\sin at}=\frac{a}{s^2+a^2}$, maka
\[
\Lap{t\sin at}= -\frac{d}{ds}\left(\frac{a}{s^2+a^2}\right)=\frac{2as}{(s^2+a^2)^2}.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Bentuk dengan $1/t$: Tidak Selalu Ada}
Ekspresi yang memuat $1/t$ harus diperiksa dekat $t=0$.
\begin{alertblock}{Contoh yang tidak konvergen}
\[
\Lap{\frac{e^{6t}}{t}}=\int_0^\infty e^{-st}\frac{e^{6t}}{t}\,dt
=\int_0^\infty \frac{e^{-(s-6)t}}{t}\,dt.
\]
Dekat $t=0$, integrannya seperti $1/t$, sehingga integralnya divergen.
\end{alertblock}
\begin{block}{Pelajaran}
Sebelum memakai tabel, cek apakah bentuknya memang memiliki Transformasi Laplace biasa.
\end{block}
\end{frame}

\begin{frame}{Bentuk $1/t$ yang Valid}
Ada bentuk $1/t$ yang valid karena pembilangnya juga menuju nol.
\[
\Lap{\frac{\sin at}{t}}=\tan^{-1}\left(\frac{a}{s}\right).
\]
Mengapa tidak bermasalah di $t=0$?
\[
\lim_{t\to0}\frac{\sin at}{t}=a.
\]
\begin{exampleblock}{Contoh}
\[
\Lap{\frac{\sin 6t}{t}}=\tan^{-1}\left(\frac{6}{s}\right).
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Integral dengan Transformasi Laplace}
Dari definisi,
\[
F(s)=\Lap{f(t)}=\int_0^\infty e^{-st}f(t)\,dt.
\]
Jadi tabel Laplace bisa dipakai untuk menghitung integral tak wajar.
\begin{exampleblock}{Contoh}
Hitung
\[
\int_0^\infty e^{-2t}\left(2-5\sin4t\right)dt.
\]
Ini sama dengan $\Lap{2-5\sin4t}$ pada $s=2$.
\end{exampleblock}
\[
\frac{2}{2}-\frac{20}{2^2+16}=1-1=0.
\]
\end{frame}

\begin{frame}{Transformasi Balik: Membaca Tabel Mundur}
\begin{exampleblock}{Contoh 1}
\[
\Inv{\frac{1}{s-3}}=e^{3t}.
\]
\end{exampleblock}
\begin{exampleblock}{Contoh 2}
\[
\Inv{\frac{4}{s^2+16}}=\sin4t.
\]
\end{exampleblock}
\begin{exampleblock}{Contoh 3}
\[
\Inv{\frac{s+2}{(s+2)^2+9}}=e^{-2t}\cos3t.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Pecahan Parsial: Jika Belum Cocok Tabel}
\begin{exampleblock}{Contoh}
\[
F(s)=\frac{5}{(s+1)(s+4)}.
\]
Tulis
\[
\frac{5}{(s+1)(s+4)}=\frac{A}{s+1}+\frac{B}{s+4}.
\]
Diperoleh $A=5/3$ dan $B=-5/3$, sehingga
\[
\Inv{F(s)}=\frac{5}{3}e^{-t}-\frac{5}{3}e^{-4t}.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Transformasi Turunan dari Definisi}
Jika $Y(s)=\Lap{y(t)}$, maka
\[
\Lap{y'}=\int_0^\infty e^{-st}y'(t)dt.
\]
Dengan integral parsial:
\[
\Lap{y'}=\left[e^{-st}y(t)\right]_0^\infty+s\int_0^\infty e^{-st}y(t)dt.
\]
Jika suku di tak hingga nol,
\[
\boxed{\Lap{y'}=sY-y(0)}.
\]
\end{frame}

\begin{frame}{Turunan Kedua dan Ketiga}
\[
\boxed{\Lap{y''}=s^2Y-sy(0)-y'(0)}
\]
\[
\boxed{\Lap{y'''}=s^3Y-s^2y(0)-sy'(0)-y''(0)}
\]
\begin{block}{Pola}
Turunan menjadi perkalian oleh $s$, tetapi kondisi awal ikut masuk.
\end{block}
\end{frame}

\begin{frame}{PDB dengan Laplace: Alur Umum}
\begin{enumerate}
\item Transformasikan seluruh suku PDB.
\item Masukkan kondisi awal.
\item Kumpulkan semua suku yang memuat $Y(s)$.
\item Selesaikan aljabar untuk memperoleh $Y(s)$.
\item Ubah $Y(s)$ kembali ke $y(t)$ dengan tabel atau pecahan parsial.
\end{enumerate}
\begin{alertblock}{Hal yang sering hilang}
Jangan lupa $y(0)$ dan $y'(0)$ ketika mentransformasikan turunan.
\end{alertblock}
\end{frame}

\begin{frame}{Contoh PDB 1: Masuk Akal Dulu}
Selesaikan
\[
y'-y=t^2e^t,\qquad y(0)=0.
\]
Transformasi:
\[
(sY-0)-Y=\Lap{t^2e^t}.
\]
Dari tabel,
\[
\Lap{t^2e^t}=\frac{2}{(s-1)^3}.
\]
Maka
\[
(s-1)Y=\frac{2}{(s-1)^3},\qquad
Y=\frac{2}{(s-1)^4}.
\]
\end{frame}

\begin{frame}{Lanjutan Contoh PDB 1}
Gunakan
\[
\Lap{t^3e^t}=\frac{3!}{(s-1)^4}=\frac{6}{(s-1)^4}.
\]
Karena
\[
Y=\frac{2}{(s-1)^4}=\frac{1}{3}\frac{6}{(s-1)^4},
\]
maka
\[
\boxed{y(t)=\frac{1}{3}t^3e^t}.
\]
\begin{block}{Cek cepat}
$y'-y=t^2e^t$ dan $y(0)=0$.
\end{block}
\end{frame}

\begin{frame}{Contoh PDB 2: Memuat $\sinh$}
Selesaikan
\[
y''-y=\sinh(2t),\qquad y(0)=0,\quad y'(0)=0.
\]
Transformasi:
\[
(s^2Y)-Y=\Lap{\sinh(2t)}=\frac{2}{s^2-4}.
\]
Maka
\[
(s^2-1)Y=\frac{2}{s^2-4},
\qquad
Y=\frac{2}{(s^2-1)(s^2-4)}.
\]
\end{frame}

\begin{frame}{Lanjutan Contoh PDB 2}
Pecahan parsial:
\[
\frac{2}{(s^2-1)(s^2-4)}
=-\frac{2}{3}\frac{1}{s^2-1}+\frac{2}{3}\frac{1}{s^2-4}.
\]
Karena
\[
\Inv{\frac{1}{s^2-1}}=\sinh t,
\qquad
\Inv{\frac{1}{s^2-4}}=\frac12\sinh 2t,
\]
maka
\[
\boxed{y(t)=-\frac{2}{3}\sinh t+\frac{1}{3}\sinh 2t}.
\]
\end{frame}

\begin{frame}{Contoh PDB 3: Eksponen Kali Sinus}
Selesaikan model latihan
\[
y''+2y'+5y=e^{-t}\sin(2t),\quad y(0)=0,\quad y'(0)=1.
\]
Transformasi kiri:
\[
\Lap{y''}+2\Lap{y'}+5\Lap{y}
=(s^2Y-1)+2sY+5Y.
\]
Transformasi kanan:
\[
\Lap{e^{-t}\sin2t}=\frac{2}{(s+1)^2+4}.
\]
Sehingga
\[
(s^2+2s+5)Y-1=\frac{2}{(s+1)^2+4}.
\]
\end{frame}

\begin{frame}{Lanjutan Contoh PDB 3}
Karena
\[
s^2+2s+5=(s+1)^2+4,
\]

diperoleh
\[
Y=\frac{1}{(s+1)^2+4}+\frac{2}{\big((s+1)^2+4\big)^2}.
\]
Bagian pertama langsung:
\[
\Inv{\frac{1}{(s+1)^2+4}}=\frac12 e^{-t}\sin2t.
\]
Bagian kedua dapat dicari memakai tabel lengkap atau sifat turunan terhadap parameter. Di sini yang penting adalah alur transformasi sampai bentuk $Y(s)$.
\end{frame}

\begin{frame}{Latihan A: Transformasi Langsung}
Hitung Transformasi Laplace berikut.
\begin{enumerate}
\item $f(t)=7e^{3t}$
\item $f(t)=2-5\sin(4t)$
\item $f(t)=e^{-2t}\cos(5t)$
\item $f(t)=3e^t\sin(2t)$
\item $f(t)=t^2e^t$
\item $f(t)=\sinh(3t)$
\end{enumerate}
\end{frame}

\begin{frame}{Latihan B: Cek Ada atau Tidak}
Tentukan apakah transformasi Laplace biasa ada. Jika ada, hitung hasilnya.
\begin{enumerate}
\item $\displaystyle f(t)=\frac{e^{6t}}{t}$
\item $\displaystyle f(t)=\frac{\sin 6t}{t}$
\item $\displaystyle f(t)=\frac{1-e^{-4t}}{t}$
\item $\displaystyle f(t)=e^{-t}\frac{\sin 3t}{t}$
\end{enumerate}
\begin{alertblock}{Petunjuk}
Periksa perilaku fungsi saat $t\to0$ sebelum memakai tabel.
\end{alertblock}
\end{frame}

\begin{frame}{Latihan C: Transformasi Balik}
Tentukan fungsi asal dari bentuk berikut.
\begin{enumerate}
\item $\displaystyle F(s)=\frac{1}{s-4}$
\item $\displaystyle F(s)=\frac{6}{(s+2)^4}$
\item $\displaystyle F(s)=\frac{4}{s^2+16}$
\item $\displaystyle F(s)=\frac{s+1}{(s+1)^2+9}$
\item $\displaystyle F(s)=\frac{3}{(s-2)^2+9}$
\item $\displaystyle F(s)=\frac{10}{(s+1)(s+4)}$
\end{enumerate}
\end{frame}

\begin{frame}{Latihan D: PDB Bertahap}
Selesaikan dengan Transformasi Laplace.
\begin{enumerate}
\item $y'-2y=3e^{2t},\quad y(0)=1$.
\item $y'+y=t^2e^{-t},\quad y(0)=0$.
\item $y''+4y=\cos(2t),\quad y(0)=0,\ y'(0)=1$.
\item $y''-y=\sinh(3t),\quad y(0)=0,\ y'(0)=0$.
\item $y''+2y'+5y=e^{-t}\cos(2t),\quad y(0)=1,\ y'(0)=0$.
\end{enumerate}
\end{frame}

\begin{frame}{Ringkasan Pertemuan 4}
\begin{enumerate}
\item Definisi Laplace berasal dari integral tak wajar.
\item Tabel mempercepat transformasi dan transformasi balik.
\item Eksponen kali sinus/cosinus diselesaikan dengan pergeseran pada $s$.
\item Bentuk $1/t$ harus dicek konvergensinya.
\item PDB diselesaikan dengan mengubah turunan menjadi bentuk aljabar dalam $Y(s)$.
\end{enumerate}
\begin{block}{Latihan utama}
Biasakan menulis langkah lengkap: transformasi, substitusi kondisi awal, aljabar $Y(s)$, invers Laplace.
\end{block}

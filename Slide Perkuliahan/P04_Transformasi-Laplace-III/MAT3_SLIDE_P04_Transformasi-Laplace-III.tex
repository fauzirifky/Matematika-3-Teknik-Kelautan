\begin{frame}{Orientasi Pertemuan 4}
\begin{block}{Fokus}
Pertemuan ini diarahkan untuk memperkuat kesiapan menghadapi jenis soal Transformasi Laplace yang biasa muncul pada latihan dan UTS: transformasi fungsi, transformasi balik, operasi lanjutan, serta penyelesaian PDB.
\end{block}
\begin{enumerate}
\item rekap definisi dan tabel dasar,
\item operasi lanjutan: shift, perkalian dengan $t^n$, pembagian oleh $t$,
\item contoh integral yang menghasilkan logaritma atau arctan,
\item penyelesaian PDB dengan ruas yang memuat $\sinh t$ dan $t^2 e^t$.
\end{enumerate}
\end{frame}

\begin{frame}{Definisi Transformasi Laplace}
\[
F(s)=\Lap{f(t)}=\int_0^{\infty} e^{-st} f(t)\,dt.
\]
\begin{block}{Makna}
Fungsi pada domain waktu $f(t)$ dipetakan menjadi fungsi baru $F(s)$ pada domain Laplace. Pada domain ini, operasi diferensial sering berubah menjadi operasi aljabar yang lebih mudah ditangani.
\end{block}
\boasnote
\end{frame}

\begin{frame}{Tabel Dasar yang Harus Dikuasai}
\small
\begin{center}
\begin{tabular}{>{\centering\arraybackslash}m{.38\textwidth} >{\centering\arraybackslash}m{.44\textwidth}}
\toprule
$f(t)$ & $F(s)=\Lap{f(t)}$\\
\midrule
$1$ & $\dfrac{1}{s}$\\[2mm]

$e^{at}$ & $\dfrac{1}{s-a}$\\[2mm]

\bottomrule
\end{tabular}
\end{center}
\end{frame}

\begin{frame}{Rekap Contoh Transformasi Dasar}
Tentukan transformasi Laplace dari:
\begin{enumerate}
\item $2-5\sin(4t)$
\item $e^{3t}\cos(2t)$
\item $t^2 e^t$
\item $\sinh t$
\end{enumerate}
\vspace{1mm}
\begin{exampleblock}{Jawaban singkat}
\[
\Lap{2-5\sin(4t)}=\frac{2}{s}-\frac{20}{s^2+16}
\]
\[
\Lap{e^{3t}\cos 2t}=\frac{s-3}{(s-3)^2+4},\qquad \Lap{t^2e^t}=\frac{2}{(s-1)^3}
\]
\[
\Lap{\sinh t}=\frac{1}{s^2-1}
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Operasi yang Boleh dan Tidak Boleh}
\begin{block}{Boleh}
\[
\Lap{af(t)+bg(t)}=aF(s)+bG(s)
\]
\[
\Lap{e^{at}f(t)}=F(s-a)
\]
\[
\Lap{tf(t)}=-\frac{d}{ds}F(s)
\]
\end{block}
\begin{alertblock}{Tidak boleh disederhanakan seperti ini}
\[
\Lap{f(t)g(t)} \neq F(s)G(s)
\]
secara umum.
\end{alertblock}
\end{frame}

\begin{frame}{Perkalian oleh $t$ dan $t^n$}
Jika $F(s)=\Lap{f(t)}$, maka
\[
\Lap{t f(t)}=-\frac{dF}{ds},
\qquad
\Lap{t^n f(t)}=(-1)^n\frac{d^nF}{ds^n}.
\]
\begin{exampleblock}{Contoh}
Dari $\Lap{e^t}=\dfrac{1}{s-1}$, diperoleh
\[
\Lap{t e^t}=-\frac{d}{ds}\left(\frac{1}{s-1}\right)=\frac{1}{(s-1)^2}
\]
dan
\[
\Lap{t^2 e^t}=\frac{d^2}{ds^2}\left(\frac{1}{s-1}\right)=\frac{2}{(s-1)^3}.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Apakah $1/t$ Bisa Ditulis sebagai $t^{-1}$?}
Bisa saja secara aljabar kita menulis
\[
\frac{1}{t}=t^{-1}.
\]
Namun untuk Transformasi Laplace biasa,
\[
\Lap{t^n}=\frac{\Gamma(n+1)}{s^{n+1}}
\]
berlaku untuk $n>-1$.
\begin{alertblock}{Akibatnya}
Untuk $n=-1$, integral
\[
\Lap{1/t}=\int_0^{\infty}\frac{e^{-st}}{t}\,dt
\]
\textbf{divergen} di sekitar $t=0$.
\end{alertblock}
Jadi $\Lap{1/t}$ tidak ada dalam pengertian biasa.
\end{frame}

\begin{frame}{Pembagian oleh $t$}
Jika $F(s)=\Lap{f(t)}$, maka berlaku aturan
\[
\Lap{\frac{f(t)}{t}}=\int_s^{\infty} F(u)\,du,
\]
asalkan integralnya konvergen.
\begin{block}{Catatan penting}
Aturan ini \textbf{bukan} berarti kita bebas mengubah $1/t$ menjadi $t^{-1}$ lalu memakai rumus pangkat biasa. Yang dipakai adalah aturan integral di atas.
\end{block}
\end{frame}

\begin{frame}{Contoh yang Salah jika Diperlakukan Terlalu Cepat}
Misalkan ingin mencari
\[
\Lap{\frac{e^{6t}}{t}}.
\]
Dari $\Lap{e^{6t}}=\dfrac{1}{s-6}$, aturan pembagian oleh $t$ memberi
\[
\Lap{\frac{e^{6t}}{t}}=\int_s^{\infty}\frac{1}{u-6}\,du.
\]
Integral ini divergen, sehingga contoh ini \textbf{tidak cocok} untuk Laplace biasa.
\begin{alertblock}{Pesan utama}
Tidak semua bentuk $f(t)/t$ mempunyai transformasi Laplace yang terdefinisi.
\end{alertblock}
\end{frame}

\begin{frame}{Contoh yang Baik: Selisih Dua Eksponensial Dibagi $t$}
Cari
\[
\Lap{\frac{e^{-2t}-e^{-5t}}{t}}.
\]
Karena
\[
\Lap{e^{-2t}}=\frac{1}{s+2},\qquad \Lap{e^{-5t}}=\frac{1}{s+5},
\]
maka
\[
\Lap{\frac{e^{-2t}-e^{-5t}}{t}}
=
\int_s^{\infty}\left(\frac{1}{u+2}-\frac{1}{u+5}\right)du.
\]
\end{frame}

\begin{frame}{Lanjutan Contoh Pembagian oleh $t$}
\[
\int_s^{\infty}\left(\frac{1}{u+2}-\frac{1}{u+5}\right)du
=
\left[\ln(u+2)-\ln(u+5)\right]_s^{\infty}.
\]
Bagian limit di tak hingga bernilai $0$, sehingga
\[
\Lap{\frac{e^{-2t}-e^{-5t}}{t}}=
\ln\left(\frac{s+5}{s+2}\right).
\]
\begin{exampleblock}{Pelajaran}
Pembagian oleh $t$ sering melahirkan bentuk logaritma.
\end{exampleblock}
\end{frame}

\begin{frame}{Membuktikan Rumus dari Rumus Dasar}
Contoh: buktikan
\[
\Lap{t^2 e^t}=\frac{2}{(s-1)^3}.
\]
Mulai dari rumus dasar
\[
\Lap{e^t}=\frac{1}{s-1}.
\]
Karena
\[
\Lap{t^2 f(t)}=\frac{d^2}{ds^2}F(s),
\]
maka
\[
\Lap{t^2 e^t}=\frac{d^2}{ds^2}\left(\frac{1}{s-1}\right)=\frac{2}{(s-1)^3}.
\]
\end{frame}

\begin{frame}{Contoh Transformasi Balik}
Tentukan invers Laplace dari:
\begin{enumerate}
\item $\dfrac{2}{(s-1)^3}$
\item $\dfrac{1}{s^2-1}$
\item $\ln\left(\dfrac{s+5}{s+2}\right)$
\end{enumerate}
\vspace{1mm}
\begin{exampleblock}{Jawaban}
\[
\invLap{\frac{2}{(s-1)^3}}=t^2 e^t,
\qquad
\invLap{\frac{1}{s^2-1}}=\sinh t
\]
dan dari hasil sebelumnya
\[
\invLap{\ln\left(\frac{s+5}{s+2}\right)}=\frac{e^{-2t}-e^{-5t}}{t}.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{PDB 1: Ruas Memuat $\sinh t$}
Selesaikan
\[
y''-y=\sinh t,
\qquad y(0)=0,
\qquad y'(0)=1.
\]
Transformasi Laplace memberi
\[
(s^2Y-1)-Y=\frac{1}{s^2-1}.
\]
Maka
\[
(s^2-1)Y=1+\frac{1}{s^2-1}.
\]
\end{frame}

\begin{frame}{Lanjutan PDB 1}
\[
Y=\frac{1}{s^2-1}+\frac{1}{(s^2-1)^2}.
\]
Dengan pecahan parsial,
\[
\frac{1}{(s^2-1)^2}
=\frac{1}{4}\left(\frac{1}{(s-1)^2}+\frac{1}{(s+1)^2}\right)-\frac{1}{4}\left(\frac{1}{s-1}-\frac{1}{s+1}\right).
\]
Sehingga invers Laplace memberi bentuk solusi eksplisit.\vspace{1mm}

\begin{exampleblock}{Hasil}
\[
y(t)=\sinh t+\frac14\,t e^t+\frac14\,t e^{-t}-\frac12\sinh t
=\frac12\sinh t+\frac14 t(e^t+e^{-t}).
\]
\end{exampleblock}
\end{frame}

\begin{frame}{PDB 2: Ruas Memuat $t^2e^t$}
Selesaikan
\[
y'+2y=t^2e^t,
\qquad y(0)=0.
\]
Transformasi Laplace memberi
\[
(sY-0)+2Y=\frac{2}{(s-1)^3}.
\]
Jadi
\[
Y=\frac{2}{(s+2)(s-1)^3}.
\]
Langkah berikutnya adalah pecahan parsial.
\end{frame}

\begin{frame}{Pecahan Parsial untuk PDB 2}
Tuliskan
\[
\frac{2}{(s+2)(s-1)^3}
=
\frac{A}{s+2}+\frac{B}{s-1}+\frac{C}{(s-1)^2}+\frac{D}{(s-1)^3}.
\]
Diperoleh
\[
A=-\frac{2}{27},\qquad B=\frac{2}{27},\qquad C=-\frac{2}{9},\qquad D=\frac{2}{3}.
\]
Maka
\[
Y=-\frac{2}{27}\frac{1}{s+2}+\frac{2}{27}\frac{1}{s-1}-\frac{2}{9}\frac{1}{(s-1)^2}+\frac{2}{3}\frac{1}{(s-1)^3}.
\]
\end{frame}

\begin{frame}{Hasil PDB 2}
Ambil invers Laplace satu per satu:
\[
y(t)=-\frac{2}{27}e^{-2t}+\frac{2}{27}e^t-\frac{2}{9}te^t+\frac13 t^2 e^t.
\]
\begin{block}{Poin penting}
Soal seperti ini memaksa kita menguasai sekaligus:
\begin{itemize}
\item shift eksponensial,
\item transformasi $t^2f(t)$,
\item pecahan parsial,
\item transformasi balik.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Latihan}
Tentukan transformasi Laplace dari:
\begin{enumerate}
\item $2-5\sin(4t)$
\item $e^{2t}\sin(3t)$
\item $t^2e^t$
\item $\dfrac{e^{-t}-e^{-4t}}{t}$
\item $\sinh(2t)$
\end{enumerate}
\end{frame}

\begin{frame}{Latihan}
Tentukan invers Laplace dari:
\begin{enumerate}
\item $\dfrac{s-2}{(s-2)^2+9}$
\item $\dfrac{3}{(s-1)^2+9}$
\item $\dfrac{2}{(s-1)^3}$
\item $\dfrac{2}{s}-\dfrac{20}{s^2+16}$
\item $\ln\left(\dfrac{s+4}{s+1}\right)$
\end{enumerate}
\end{frame}

\begin{frame}{Latihan}
Selesaikan dengan Transformasi Laplace:
\begin{enumerate}
\item $y'+3y=t^2e^t,\quad y(0)=1$
\item $y''-4y=\sinh(2t),\quad y(0)=0,\ y'(0)=0$
\item $y''+2y'+5y=e^{-t}\cos 2t,\quad y(0)=0,\ y'(0)=1$
\end{enumerate}
\end{frame}

\begin{frame}{Ringkasan}
\begin{enumerate}
\item $1/t$ boleh ditulis $t^{-1}$ secara aljabar, tetapi \textbf{tidak otomatis} boleh memakai rumus pangkat biasa pada Laplace.
\item Operasi hasil kali tidak berubah menjadi hasil kali transformasi secara umum.
\item Untuk $f(t)/t$, gunakan aturan
\[
\Lap{f(t)/t}=\int_s^{\infty}F(u)\,du.
\]
\item Contoh penting minggu ini: $t^2e^t$, $\sinh t$, dan bentuk logaritma dari pembagian oleh $t$.
\item Penyelesaian PDB dengan Laplace menuntut penguasaan tabel, operasi, dan pecahan parsial.
\end{enumerate}
\end{frame}
